cmake_minimum_required(VERSION 3.28)
project(ValMAXLauncherSupport VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "ValMAX launcher support requires Windows x64 MSVC")
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
add_library(valmax_activation STATIC activation.cpp activation.h)
target_compile_definitions(valmax_activation PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(valmax_activation PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(valmax_activation PUBLIC user32)
target_compile_options(valmax_activation PRIVATE /W4 /WX /permissive- /utf-8)
enable_testing()
add_executable(valmax_activation_test ../tests/launcher/activation_test.cpp)
target_link_libraries(valmax_activation_test PRIVATE valmax_activation)
target_compile_options(valmax_activation_test PRIVATE /W4 /WX /permissive- /utf-8)
add_test(NAME activation_algorithm COMMAND valmax_activation_test)
add_executable(valmax_activation_metadata_test ../tests/launcher/activation_metadata_test.cpp)
target_link_libraries(valmax_activation_metadata_test PRIVATE valmax_activation)
target_compile_options(valmax_activation_metadata_test PRIVATE /W4 /WX /permissive- /utf-8)
add_test(NAME activation_owned_metadata COMMAND valmax_activation_metadata_test)

set(VALMAX_PREFLIGHT_RUNTIME_ROOT "" CACHE PATH "Immutable owned offline fixture to embed; not an installed client")
set(VALMAX_BUILD_PYTHON "" CACHE FILEPATH "Build-only Python with pefile and inspected build_mod dependencies")
if(VALMAX_PREFLIGHT_RUNTIME_ROOT AND VALMAX_BUILD_PYTHON)
    add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/preflight_pins.generated.h" "${CMAKE_CURRENT_BINARY_DIR}/preflight-pins.json"
        COMMAND "${VALMAX_BUILD_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/preflight_generate_pins.py"
            --runtime-root "${VALMAX_PREFLIGHT_RUNTIME_ROOT}"
            --output "${CMAKE_CURRENT_BINARY_DIR}/preflight_pins.generated.h"
            --proof "${CMAKE_CURRENT_BINARY_DIR}/preflight-pins.json"
        DEPENDS preflight_generate_pins.py "${VALMAX_PREFLIGHT_RUNTIME_ROOT}/../build-proof.json"
            "${VALMAX_PREFLIGHT_RUNTIME_ROOT}/mod-manifest.json"
        VERBATIM)
    add_executable(valmax_preflight preflight_main.cpp "${CMAKE_CURRENT_BINARY_DIR}/preflight_pins.generated.h")
    target_link_libraries(valmax_preflight PRIVATE valmax_preflight_support)
    target_include_directories(valmax_preflight PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    target_compile_options(valmax_preflight PRIVATE /W4 /WX /permissive- /utf-8)
endif()
