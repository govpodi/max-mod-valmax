cmake_minimum_required(VERSION 3.28)
project(ValMAXTransferApp VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Windows x64 MSVC required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
find_package(Qt6 6.11.2 EXACT REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2)
set(VALMAX_CORE_LIBRARY "" CACHE FILEPATH "Pinned Rust core DLL import library")
if(NOT VALMAX_CORE_LIBRARY OR NOT EXISTS "${VALMAX_CORE_LIBRARY}")
    message(FATAL_ERROR "VALMAX_CORE_LIBRARY must name the frozen core import library")
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../host_callbacks.cmake")
include("${CMAKE_CURRENT_LIST_DIR}/../../common/startup_context.cmake")
set(app_sources backend.cpp backend.h worker.cpp worker.h controller.cpp controller.h)
function(transfer_target target)
    target_link_libraries(${target} PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2
        valmax_transfer_host "${VALMAX_CORE_LIBRARY}" shell32 ole32 uuid)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
    target_compile_definitions(${target} PRIVATE QT_NO_CAST_FROM_ASCII QT_NO_CAST_TO_ASCII)
    qt_add_resources(${target} transfer_ui PREFIX "/valmax-transfer" FILES Main.qml)
endfunction()
add_executable(valmax-transfer WIN32 main.cpp ${app_sources})
transfer_target(valmax-transfer)
target_link_libraries(valmax-transfer PRIVATE valmax_startup_context user32)
option(VALMAX_TRANSFER_APP_BUILD_TESTS "Build isolated fake-backend application tests" OFF)
if(VALMAX_TRANSFER_APP_BUILD_TESTS)
    add_executable(valmax_transfer_app_test tests/app_test.cpp ${app_sources})
    target_compile_definitions(valmax_transfer_app_test PRIVATE VALMAX_TRANSFER_APP_TESTING)
    transfer_target(valmax_transfer_app_test)
endif()
