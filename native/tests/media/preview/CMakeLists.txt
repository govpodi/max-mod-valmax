cmake_minimum_required(VERSION 3.28)
project(ValMAXPreviewProviderProof LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(MEDIA "${CMAKE_CURRENT_LIST_DIR}/../../../media")
find_package(Qt6 6.11.2 EXACT REQUIRED COMPONENTS Core Gui Qml Quick)
add_library(valmax_media_preview STATIC "${MEDIA}/preview_provider.cpp" "${MEDIA}/preview_provider.h")
target_include_directories(valmax_media_preview PUBLIC "${MEDIA}")
target_link_libraries(valmax_media_preview PUBLIC Qt6::Gui Qt6::Quick Qt6::Qml)
target_compile_definitions(valmax_media_preview PRIVATE QT_NO_CAST_FROM_ASCII QT_NO_CAST_TO_ASCII)
add_executable(valmax_preview_provider_test ../preview_provider_test.cpp)
target_link_libraries(valmax_preview_provider_test PRIVATE valmax_media_preview)
if(MSVC)
  target_compile_options(valmax_media_preview PRIVATE /W4 /WX /permissive- /utf-8)
  target_compile_options(valmax_preview_provider_test PRIVATE /W4 /WX /permissive- /utf-8)
endif()
