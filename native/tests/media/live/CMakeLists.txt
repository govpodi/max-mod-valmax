cmake_minimum_required(VERSION 3.28)
project(ValMAXMediaLiveProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(VALMAX_MEDIA_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(../../../media media)
add_executable(valmax_media_live_probe ../live_probe.cpp)
target_link_libraries(valmax_media_live_probe PRIVATE valmax_media_controller valmax_media_thumbnail)
target_compile_definitions(valmax_media_live_probe PRIVATE QT_NO_CAST_FROM_ASCII QT_NO_CAST_TO_ASCII)
target_compile_options(valmax_media_live_probe PRIVATE /W4 /WX /permissive- /utf-8)
