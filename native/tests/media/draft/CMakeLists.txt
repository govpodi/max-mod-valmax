cmake_minimum_required(VERSION 3.28)
project(ValMAXDraftContractProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
find_package(Qt6 6.11.2 EXACT REQUIRED COMPONENTS Core Qml)
add_library(valmax_draft_contract_probe SHARED draft_probe.cpp)
target_link_libraries(valmax_draft_contract_probe PRIVATE Qt6::Core Qt6::Qml)
target_compile_definitions(valmax_draft_contract_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(valmax_draft_contract_probe PRIVATE /W4 /WX /permissive- /utf-8)
