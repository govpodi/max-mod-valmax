cmake_minimum_required(VERSION 3.28)
project(ValMAXClosedClients LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Windows x64 MSVC required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
include("${CMAKE_CURRENT_LIST_DIR}/../../transfer/closed_clients.cmake")
add_executable(valmax_closed_clients_test closed_clients_test.cpp)
target_link_libraries(valmax_closed_clients_test PRIVATE valmax_closed_clients)
target_compile_options(valmax_closed_clients_test PRIVATE /W4 /WX /permissive- /utf-8)
enable_testing()
add_test(NAME closed_clients COMMAND valmax_closed_clients_test)
