cmake_minimum_required(VERSION 3.28)
project(ValMAXTransferHost LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Windows x64 MSVC required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
include("${CMAKE_CURRENT_LIST_DIR}/../../../transfer/host_callbacks.cmake")
# Compile production bytes too; the test seam is never in this library.
add_library(valmax_transfer_host_synthetic STATIC ../../../transfer/host_callbacks.cpp)
target_compile_definitions(valmax_transfer_host_synthetic PUBLIC VALMAX_TRANSFER_TESTING)
target_link_libraries(valmax_transfer_host_synthetic PUBLIC valmax_closed_clients)
target_compile_options(valmax_transfer_host_synthetic PRIVATE /W4 /WX /permissive- /utf-8)
add_executable(valmax_transfer_host_test host_callbacks_test.cpp)
target_link_libraries(valmax_transfer_host_test PRIVATE valmax_transfer_host_synthetic)
target_compile_options(valmax_transfer_host_test PRIVATE /W4 /WX /permissive- /utf-8)
enable_testing()
add_test(NAME transfer_host_callbacks COMMAND valmax_transfer_host_test)
