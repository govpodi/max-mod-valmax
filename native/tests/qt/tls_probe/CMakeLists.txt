cmake_minimum_required(VERSION 3.28)
project(ValmaxOrdinaryTlsProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 6.11.2 EXACT REQUIRED COMPONENTS Core Network)
add_executable(valmax_tls_probe main.cpp)
target_compile_definitions(valmax_tls_probe PRIVATE QT_NO_CAST_FROM_ASCII QT_NO_CAST_TO_ASCII)
target_compile_options(valmax_tls_probe PRIVATE /W4 /WX /permissive- /utf-8)
target_link_libraries(valmax_tls_probe PRIVATE Qt6::Core Qt6::Network)
set_target_properties(valmax_tls_probe PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
