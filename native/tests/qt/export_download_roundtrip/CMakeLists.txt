cmake_minimum_required(VERSION 3.28)
project(ValMAXExportDownloadRoundtrip LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Windows x64 MSVC test only")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.11.2 EXACT REQUIRED COMPONENTS Core Network)
set(VALMAX_CORE_LIBRARY "" CACHE FILEPATH "Frozen real Rust import library")
if(NOT VALMAX_CORE_LIBRARY OR NOT EXISTS "${VALMAX_CORE_LIBRARY}")
    message(FATAL_ERROR "A frozen real Rust library is required")
endif()
get_filename_component(VALMAX_NATIVE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(valmax_export_download_roundtrip ../export_download_roundtrip_test.cpp
    "${VALMAX_NATIVE_ROOT}/qt/export_downloader.cpp" "${VALMAX_NATIVE_ROOT}/qt/export_downloader.h")
target_include_directories(valmax_export_download_roundtrip PRIVATE "${VALMAX_NATIVE_ROOT}/qt" "${VALMAX_NATIVE_ROOT}/core")
target_compile_definitions(valmax_export_download_roundtrip PRIVATE VALMAX_HAS_RUST_CORE=1 VALMAX_EXPORT_DOWNLOAD_TESTING
    QT_NO_CAST_FROM_ASCII QT_NO_CAST_TO_ASCII)
target_compile_options(valmax_export_download_roundtrip PRIVATE /W4 /WX /permissive- /utf-8)
target_link_libraries(valmax_export_download_roundtrip PRIVATE Qt6::Core Qt6::Network "${VALMAX_CORE_LIBRARY}")
set_target_properties(valmax_export_download_roundtrip PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
