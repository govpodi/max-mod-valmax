cmake_minimum_required(VERSION 3.28)
project(ValMAXSyntheticBootstrap LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Synthetic bootstrap requires Windows x64 MSVC")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
find_package(Qt6 6.11.2 EXACT REQUIRED COMPONENTS Core Gui)
add_library(valmaxbootstrapplugin MODULE plugin.cpp markers.h plugin.json)
target_link_libraries(valmaxbootstrapplugin PRIVATE Qt6::Core Qt6::Gui)
set_target_properties(valmaxbootstrapplugin PROPERTIES PREFIX "")
add_executable(valmax_bootstrap_harness harness.cpp markers.h)
target_link_libraries(valmax_bootstrap_harness PRIVATE Qt6::Core Qt6::Gui)
foreach(target valmaxbootstrapplugin valmax_bootstrap_harness)
  target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX QT_NO_CAST_FROM_ASCII QT_NO_CAST_TO_ASCII)
  target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
endforeach()
