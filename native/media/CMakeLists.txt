cmake_minimum_required(VERSION 3.28)
project(ValMAXMediaProtocol LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
find_package(Qt6 6.11.2 EXACT REQUIRED COMPONENTS Core)
add_library(valmax_media_protocol STATIC protocol.cpp protocol.h html_entities.inc)
target_include_directories(valmax_media_protocol PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(valmax_media_protocol PUBLIC Qt6::Core)
target_compile_definitions(valmax_media_protocol PRIVATE QT_NO_CAST_FROM_ASCII QT_NO_CAST_TO_ASCII)
if(MSVC)
  target_compile_options(valmax_media_protocol PRIVATE /W4 /WX /permissive- /utf-8)
endif()
option(VALMAX_MEDIA_TESTS "Build synthetic pure media tests" ON)
if(VALMAX_MEDIA_TESTS)
  enable_testing()
  add_executable(valmax_media_protocol_test ../tests/media/protocol_test.cpp)
  target_link_libraries(valmax_media_protocol_test PRIVATE valmax_media_protocol)
  if(MSVC)
    target_compile_options(valmax_media_protocol_test PRIVATE /W4 /WX /permissive- /utf-8)
  endif()
  add_test(NAME media_protocol COMMAND valmax_media_protocol_test)
endif()
include(scheduler.cmake OPTIONAL)
include(result_cache.cmake OPTIONAL)
include(transport.cmake OPTIONAL)
include(controller.cmake OPTIONAL)
option(VALMAX_MEDIA_THUMBNAILS "Build bounded in-memory thumbnail decoder" ON)
if(VALMAX_MEDIA_THUMBNAILS)
  include(thumbnail.cmake)
  include(decode_queue.cmake)
  include(preview_provider.cmake)
  include(engine_context.cmake)
endif()
